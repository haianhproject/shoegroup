    if (selBrand.value) params.append('brand', selBrand.value)
    if (inStockOnly.value) params.append('inStock', 'true')
    if (minPrice.value) params.append('minPrice', minPrice.value)
    if (maxPrice.value) params.append('maxPrice', maxPrice.value)

    if (selSports.value.length) params.append('sports', selSports.value.join(','))
    if (selColors.value.length) params.append('colors', selColors.value.join(','))

    if (selMaterials.value.length) params.append('materials', selMaterials.value.join(','))

    // Ưu tiên gọi API v2 có phân trang

    if (data.items && Array.isArray(data.items)) {

const parseURLQuery = () => {
  const q = route.query
  search.value = q.search || ''
  selCategory.value = q.category ? Number(q.category) : null
  selBrand.value = q.brand ? Number(q.brand) : null
  inStockOnly.value = q.inStock === 'true'
  minPrice.value = q.minPrice || ''
  maxPrice.value = q.maxPrice || ''
  selSports.value = q.sports ? q.sports.split(',') : []
  selColors.value = q.colors ? q.colors.split(',') : []
  selSizes.value = q.sizes ? q.sizes.split(',') : []
  selMaterials.value = q.materials ? q.materials.split(',').map(Number) : []
  sortBy.value = q.sort || 'featured'
  page.value = Number(q.page) || 1
}

// Đẩy trạng thái Bộ Lọc lên URL
const updateURL = (resetPage = false) => {
  if (resetPage) page.value = 1

  const query = {}
  if (search.value) query.search = search.value
  if (selCategory.value) query.category = selCategory.value
  if (selBrand.value) query.brand = selBrand.value
  if (inStockOnly.value) query.inStock = 'true'
  if (minPrice.value) query.minPrice = minPrice.value
  if (maxPrice.value) query.maxPrice = maxPrice.value
  if (selSports.value.length) query.sports = selSports.value.join(',')
  if (selColors.value.length) query.colors = selColors.value.join(',')
  if (selSizes.value.length) query.sizes = selSizes.value.join(',')
  if (selMaterials.value.length) query.materials = selMaterials.value.join(',')
  if (sortBy.value !== 'featured') query.sort = sortBy.value
  if (page.value > 1) query.page = page.value

  router.push({ path: '/products', query })
}

// Toggle phần tử trong mảng lọc
const toggle = (arr, val) => {
  const i = arr.indexOf(val)
  if (i === -1) arr.push(val); else arr.splice(i, 1)
  updateURL(true)
}
